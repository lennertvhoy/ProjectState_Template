---
name: "close-slice"
gate_level: 2
evidence_max: 8
cheapest_proof: "Authoritative local quality gate plus exact-head remote finalizer"
escalate_when: "Release or template migration requires level 3 with CI proof"
description: "Execute full slice closure: audits, commit, state updates, evidence, quality gates"
when_to_use:
  - "Closing an implementation slice"
  - "User-facing or operator-facing changes ready for acceptance"
  - "Before marking any work as closure-grade"
required_inputs:
  - "Slice ID or scope description"
  - "Current branch with changes"
  - "Evidence collected during implementation"
step_by_step:
  - name: "Run authoritative local quality gate"
    command: "python3 scripts/statedd_quality_gate.py --gate-level 2"
    expected: "Exit 0 after all applicable suites and local checks run"
    failure: "Fix every aggregated failure and re-run the same command"
    notes: "Parallel agents must provision an isolated worktree before implementation."
  - name: "Commit implementation proof"
    action: "Commit only classified slice work, then record that commit as the evidence proof head"
    verify: "Clean worktree and tracked evidence; no final-head self-reference is fabricated"
  - name: "Finalize state and evidence"
    action: "Mark canonical state validated_local_remote_pending, keep the backlog item active, append the local boundary to WORKLOG, hash the strict evidence pack, and commit only finalization metadata"
    verify: "The proof-to-final diff contains only allowlisted state, evidence, handoff, or metric metadata"
  - name: "Push and open or update the pull request"
    action: "Push normally; make the PR body name both the proof head and exact final PR head; never merge without explicit approval"
    verify: "PR head equals local and remote branch HEAD"
  - name: "Wait for exact-head CI"
    action: "Observe the latest GitHub Actions run for the exact final PR head"
    verify: "Required checks completed successfully for that SHA; earlier CI is not reused"
  - name: "Run remote closure finalizer"
    command: "python3 scripts/statedd_remote_closure_finalizer.py --pr-number <pr> --evidence-folder <folder>"
    expected: "Exit 0 after local HEAD, remote branch, PR body/head, tracked evidence, Actions, and merge state agree"
    failure: "Refresh PR/evidence or fix CI, push a new final head, and rerun exact-head proof"
    notes: "Its external receipt may assert exact-head CI truth. A later in-repo done/CI-verified update is a new head and requires fresh CI/finalizer proof."
  - name: "Generate final handoff"
    command: "python3 scripts/statedd_handoff.py --run-audit --test-command 'python3 scripts/statedd_quality_gate.py --gate-level 2'"
    expected: "Exit 0 and report local, pushed, PR, CI, runtime, and acceptance boundaries separately"
expected_outputs:
  - "Authoritative local quality gate passes"
  - "Exact-head remote closure finalizer passes"
  - "State and strict evidence pack are tracked and aligned"
  - "Clean worktree and CTO-pasteable handoff"
failure_cases:
  - name: "Local gate fails"
    detection: "statedd_quality_gate.py exits non-zero"
    recovery: "Fix all reported failures and rerun the authoritative command"
    evidence: "Quality-gate output"
  - name: "Remote closure not verified"
    detection: "statedd_remote_closure_finalizer.py exits non-zero"
    recovery: "Align exact heads, PR body, evidence, CI, and merge state without merging"
    evidence: "Remote finalizer output"
evidence_required:
  - "Authoritative quality-gate output"
  - "Strict evidence bundle"
  - "Runtime/browser proof only when the change type requires it"
  - "Remote finalizer output and handoff text"
exit_criteria:
  - "Local gate and exact-head remote finalizer exit 0"
  - "State files pass check_state_docs.py"
  - "Acceptance is recorded only when a human actually accepted the result"
