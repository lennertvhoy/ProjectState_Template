---
name: "close-slice"
description: "Execute full slice closure: audits, commit, state updates, evidence, quality gates"
when_to_use:
  - "Closing an implementation slice"
  - "User-facing or operator-facing changes ready for acceptance"
  - "Before marking any work as closure-grade"
required_inputs:
  - "Slice ID or scope description"
  - "Current branch with changes"
  - "Evidence collected during implementation"
step_by_step:
  - name: "Run quality gate"
    command: "python scripts/statedd_quality_gate.py"
    expected: "Exit 0 (all tests pass, static analysis clean, state updated, evidence recorded)"
    failure: "Fix failures, re-run. If blocked, record partial status in handoff."
  - name: "Run closure check"
    command: "python scripts/statedd_closure_check.py"
    expected: "Exit 0 (closure criteria met)"
    failure: "Address gaps, re-run."
  - name: "Run instruction lint"
    command: "python scripts/statedd_instruction_lint.py"
    expected: "Exit 0 (no config smells)"
    failure: "Fix smells, re-run."
  - name: "Run runtime truth check"
    command: "python scripts/statedd_runtime_truth_check.py"
    expected: "Exit 0 (runtime identity matches recorded)"
    failure: "Update runtime_identity.json or evidence, re-run."
  - name: "Run evidence type check"
    command: "python scripts/statedd_evidence_type_check.py"
    expected: "Exit 0 (evidence matches change type)"
    failure: "Collect missing evidence, re-run."
  - name: "Update state files"
    action: "Update PROJECT_STATE.yaml, STATUS.md, NEXT_ACTIONS.md, BACKLOG.md, WORKLOG.md"
    verify: "python scripts/check_state_docs.py"
  - name: "Generate handoff"
    command: "python scripts/statedd_handoff.py"
    expected: "Complete handoff text for CTO lane"
  - name: "Commit and push"
    action: "git add -A && git commit -m 'close-slice: <slice-id>' && git push"
    verify: "Clean worktree, HEAD at commit"
expected_outputs:
  - "All quality gates pass (exit 0)"
  - "State files updated and valid"
  - "Handoff generated with evidence refs"
  - "Clean git worktree"
  - "Slice marked closure-grade in BACKLOG.md"
failure_cases:
  # failure cases for close-slice skill
  - name: "Quality gate fails"
    detection: "statedd_quality_gate.py exits non-zero"
    recovery: "Fix code/tests, re-run gate"
    evidence: "Gate output logs"
  - name: "Closure check fails"
    detection: "statedd_closure_check.py exits non-zero"
    recovery: "Address missing closure criteria"
    evidence: "Closure check output"
  - name: "Instruction lint fails"
    detection: "statedd_instruction_lint.py exits non-zero"
    recovery: "Fix AGENTS.md/skill/command files"
    evidence: "Lint output"
  - name: "Runtime truth mismatch"
    detection: "statedd_runtime_truth_check.py exits non-zero"
    recovery: "Update runtime_identity.json or investigate drift"
    evidence: "Runtime proof output"
  - name: "Evidence mismatch"
    detection: "statedd_evidence_type_check.py exits non-zero"
    recovery: "Collect required evidence type"
    evidence: "Evidence check output"
  - name: "State validation fails"
    detection: "check_state_docs.py exits non-zero"
    recovery: "Fix YAML/schemas"
    evidence: "Validation output"
evidence_required:
  - "Quality gate output (stdout/stderr)"
  - "Closure check output"
  - "Runtime proof (runtime_identity.json + screenshot/log)"
  - "Evidence bundle (per EVIDENCE_README_TEMPLATE.md)"
  - "Handoff text"
exit_criteria:
  - "All gates exit 0"
  - "State files pass check_state_docs.py"
  - "Handoff complete and CTO-pasteable"
  - "Slice ID recorded in ACCEPTANCE_FREEZES.md if user-facing"
