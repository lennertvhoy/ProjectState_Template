---
name: "release-gate"
gate_level: 3
evidence_max: 8
cheapest_proof: "All level 2 gates pass plus CI proof"
escalate_when: "Never; this is the heaviest gate"
description: "Final release gate before deployment or user-facing delivery"
when_to_use:
  - "Before any deployment, release, or user-facing delivery"
  - "When the CTO handoff requests release validation"
  - "After a slice has passed close-slice but needs release-grade verification"
required_inputs:
  - "Release version/tag or slice ID"
  - "Current branch with merged or ready-to-merge changes"
  - "Runtime proof and acceptance freezes for user-facing changes"
step_by_step:
  - name: "Run quality gate"
    command: "python scripts/statedd_quality_gate.py"
    expected: "Exit 0"
    failure: "Fix blocking issues and re-run"
  - name: "Run runtime proof"
    command: "python scripts/statedd_runtime_proof.py"
    expected: "Exit 0 (runtime identity captured)"
    failure: "Check deployment endpoint or runtime environment"
  - name: "Run runtime truth check"
    command: "python scripts/statedd_runtime_truth_check.py"
    expected: "Exit 0 (recorded identity matches live runtime)"
    failure: "Rebuild and re-verify"
  - name: "Run evidence type check"
    command: "python scripts/statedd_evidence_type_check.py"
    expected: "Exit 0 (release evidence complete)"
    failure: "Collect missing evidence"
  - name: "Run remote closure finalizer"
    command: "python scripts/statedd_remote_closure_finalizer.py"
    expected: "Exit 0 (pushed PR head, PR body, evidence, GitHub Actions, and merge state agree)"
    failure: "Refresh PR body/evidence, wait for CI, fix failures, re-push, re-run"
  - name: "Verify acceptance freezes"
    action: "Confirm docs/ACCEPTANCE_FREEZES.md has entries for every user-facing milestone"
    verify: "python scripts/check_state_docs.py"
  - name: "Generate release handoff"
    command: "python scripts/statedd_handoff.py"
    expected: "Complete release handoff with version, runtime identity, evidence bundle, and rollback plan"
expected_outputs:
  - "All quality gates pass (exit 0)"
  - "Remote closure finalizer passes (exit 0)"
  - "Runtime identity proven"
  - "Release handoff complete with rollback plan"
failure_cases:
  - name: "Quality gate fails"
    detection: "statedd_quality_gate.py exits non-zero"
    recovery: "Fix blocking issue, re-run"
    evidence: "Quality gate output"
  - name: "Runtime proof fails"
    detection: "statedd_runtime_proof.py exits non-zero"
    recovery: "Check deployment endpoint"
    evidence: "Runtime proof output"
  - name: "Runtime truth mismatch"
    detection: "statedd_runtime_truth_check.py exits non-zero"
    recovery: "Rebuild and re-verify"
    evidence: "Runtime truth output"
  - name: "Evidence missing"
    detection: "statedd_evidence_type_check.py exits non-zero"
    recovery: "Collect before release"
    evidence: "Evidence check output"
  - name: "Remote closure not verified"
    detection: "statedd_remote_closure_finalizer.py exits non-zero"
    recovery: "Refresh PR body/evidence, wait for CI, fix failures, re-push, re-run"
    evidence: "Remote closure finalizer output"
  - name: "Acceptance freezes incomplete"
    detection: "Missing entries in docs/ACCEPTANCE_FREEZES.md"
    recovery: "Add missing entries"
    evidence: "Acceptance freeze log"
exit_criteria:
  - "All gates exit 0"
  - "Remote closure verified"
  - "Runtime verified"
  - "Evidence complete"
  - "Release handoff ready"
