name: "quality-gate"
gate_level: 2
evidence_max: 8
cheapest_proof: "All gate scripts exit 0 for the slice"
escalate_when: "Release requires level 3 with CI proof"
description: "Execute the full quality gate pipeline for a slice"
when_to_use:
  - "After implementation, before closure"
  - "When CTO handoff requests quality freeze"
  - "Before release or deployment"
required_inputs:
  - "Slice ID"
  - "Branch with changes"
step_by_step:
  - name: "Verify Git safety permit"
    command: "python scripts/statedd_git_safety_check.py --mode <normal_branch|worktree|clone>"
    expected: "Exit 0 with mutation_permitted=true and a schema-valid report"
    failure: "Session remains read-only; repair and explicitly restart before state/source mutation"
  - name: "Run test suite"
    command: "python -m pytest (or project test command)"
    expected: "All tests pass"
    failure: "Fix failing tests, re-run"
  - name: "Run static analysis"
    command: "project linter (ruff, mypy, etc.)"
    expected: "No errors"
    failure: "Fix lint errors, re-run"
  - name: "Run statedd_quality_gate.py"
    command: "python scripts/statedd_quality_gate.py"
    expected: "Exit 0 (tests, analysis, state, evidence all pass)"
    failure: "Check output for specific failures, address each"
  - name: "Run statedd_instruction_lint.py"
    command: "python scripts/statedd_instruction_lint.py"
    expected: "Exit 0 (no config smells)"
    failure: "Fix AGENTS.md/skill/command smells"
  - name: "Run statedd_closure_check.py"
    command: "python scripts/statedd_closure_check.py"
    expected: "Exit 0 (closure criteria met)"
    failure: "Address missing criteria"
  - name: "Run statedd_runtime_truth_check.py"
    command: "python scripts/statedd_runtime_truth_check.py"
    expected: "Exit 0 (runtime matches recorded)"
    failure: "Update runtime_identity.json"
  - name: "Run statedd_evidence_type_check.py"
    command: "python scripts/statedd_evidence_type_check.py"
    expected: "Exit 0 (evidence matches change type)"
    failure: "Collect required evidence"
  - name: "Run check_state_docs.py"
    command: "python scripts/check_state_docs.py"
    expected: "Exit 0 (doc hygiene clean)"
    failure: "Fix state/doc issues"
expected_outputs:
  - "All gate scripts exit 0"
  - "Test output"
  - "Static analysis output"
  - "State validation output"
failure_cases:
  # failure cases for quality-gate skill
  - name: "Any gate fails"
    detection: "Any quality gate script exits non-zero"
    recovery: "Address specific failure, re-run full pipeline"
    evidence: "Gate output logs"
  - name: "Tests fail"
    detection: "pytest exits non-zero"
    recovery: "Fix code, not tests (unless test bug)"
    evidence: "Test output"
  - name: "State invalid"
    detection: "check_state_docs.py or schema validation fails"
    recovery: "Fix YAML/schemas"
    evidence: "Validation output"
  - name: "Evidence missing"
    detection: "statedd_evidence_type_check.py exits non-zero"
    recovery: "Collect required evidence before proceeding"
    evidence: "Evidence check output"
evidence_required:
  - "All gate script outputs"
  - "Test results"
  - "Static analysis results"
exit_criteria:
  - "All quality gates pass (exit 0)"
  - "No outstanding failures"
  - "Ready for close-slice skill"
