---
name: "ingest-bad-event"
gate_level: 1
evidence_max: 3
cheapest_proof: "Incident file + failure scan + backlog entry + handoff all exist"
escalate_when: "P0 triggers level 2 quality freeze"
description: "Record and handle an invalid input, failure, or unexpected event as a durable incident"
when_to_use:
  - "Any unhandled error, crash, or unexpected behavior observed"
  - "User reports a bug or regression"
  - "Test reveals a failure not in backlog"
  - "Mysterious log entry or alert"
required_inputs:
  - "Event description (what happened, when, where)"
  - "Observed symptoms (logs, screenshots, stack traces)"
  - "Severity assessment (per FAILURE_TAXONOMY.md)"
step_by_step:
  - name: "Record incident"
    action: "Create file in docs/incidents/YYYYMMDD-HHMMSS-<slug>.md from INCIDENT_RESPONSE.md template"
    verify: "File exists with all required fields"
  - name: "Classify severity"
    action: "Assign P0-P4 per FAILURE_TAXONOMY.md"
    verify: "Severity field matches taxonomy"
  - name: "Create failure scan"
    action: "Create docs/failure_scans/<incident-id>.md from TEMPLATE.md"
    verify: "Pre-mortem complete with adjacent failure modes"
  - name: "Add to backlog"
    action: "Add entry to BACKLOG.md with incident link, severity, and BL-<id>"
    verify: "BACKLOG.md updated with incident reference"
  - name: "Update evidence log"
    action: "Append to docs/EVIDENCE_LOG.md with incident link"
    verify: "EVIDENCE_LOG.md has entry"
  - name: "Trigger quality freeze if P0"
    action: "If P0, keep repo/workflow mode intact and set current_state.execution_mode.mode to quality_freeze in PROJECT_STATE.yaml"
    verify: "PROJECT_STATE.yaml current execution mode = quality_freeze"
  - name: "Generate handoff"
    command: "python scripts/statedd_handoff.py"
    expected: "Handoff includes incident ref and next steps"
expected_outputs:
  - "Incident file in docs/incidents/"
  - "Failure scan in docs/failure_scans/"
  - "Backlog entry with BL-<id>"
  - "Evidence log entry"
  - "Handoff with next actions"
failure_cases:
  # failure cases for ingest-bad-event skill
  - name: "Incident file creation fails"
    detection: "Cannot write to docs/incidents/"
    recovery: "Check permissions/path; create parent dirs"
    evidence: "Error output"
  - name: "Severity unclear"
    detection: "Cannot classify per FAILURE_TAXONOMY.md"
    recovery: "Default to P2, escalate in handoff"
    evidence: "Handoff notes"
  - name: "No adjacent failures found"
    detection: "Failure scan finds no related modes"
    recovery: "Record 'none identified' in scan"
    evidence: "Failure scan file"
evidence_required:
  - "Incident file path"
  - "Failure scan path"
  - "Backlog ID"
  - "Handoff text"
exit_criteria:
  - "Incident recorded with severity"
  - "Failure scan complete"
  - "Backlog updated"
  - "Handoff ready for CTO"
